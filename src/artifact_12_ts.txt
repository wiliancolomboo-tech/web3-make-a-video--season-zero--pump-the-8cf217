import "dotenv/config";
import express from "express";
import { health } from "./solana.js";
import { notify } from "./notify.js";
import { Evidence, proofHash } from "./verify.js";

const app = express();
app.use(express.json({ limit: "1mb" }));

app.get("/healthz", async (_req, res) => {
  try {
    res.json(await health());
  } catch {
    res.status(503).json({ ok: false });
  }
});

app.post("/v1/proof/hash", async (req, res) => {
  try {
    const evidence = req.body as Evidence;

    if (
      !evidence.campaignId ||
      !Number.isInteger(evidence.questId) ||
      !evidence.wallet ||
      !evidence.evidenceUrl ||
      !evidence.provider ||
      !evidence.providerObjectId ||
      !evidence.observedAt
    ) {
      return res.status(400).json({ error: "invalid evidence payload" });
    }

    const digest = Buffer.from(proofHash(evidence)).toString("hex");

    await notify(
      `Proof prepared for wallet ${evidence.wallet}, quest ${evidence.questId}`
    );

    return res.json({
      hash: digest,
      algorithm: "sha256",
      canonical: true,
      nextAction: "submit_proof"
    });
  } catch (error) {
    return res.status(500).json({
      error: error instanceof Error ? error.message : "unknown error"
    });
  }
});

app.post("/v1/notify", async (req, res) => {
  const { message, email } = req.body as {
    message?: string;
    email?: string;
  };

  if (!message || message.length > 4000) {
    return res.status(400).json({ error: "invalid message" });
  }

  await notify(message, email);
  return res.status(202).json({ accepted: true });
});

const port = Number(process.env.PORT ?? 8080);
app.listen(port, () => {
  console.log(`ProofPool worker listening on :${port}`);
});
